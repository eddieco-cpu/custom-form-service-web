async function fixture(status = 'published', rating = false, clipboard = false) {

  const page = await context.newPage();
  const s = { dto: dto(status, rating), writeMode: 'success', requests: [], errors: [], diagnostics: [], responses: [], failures: [], unexpected: [] };

  page.on('console', m => { if (m.type() === 'error' || m.text().includes('[Vue warn]')) s.diagnostics.push({ text: m.text(), url: m.location().url }); });
  page.on('response', r => { if (r.status() >= 400) s.responses.push({ method: r.request().method(), url: r.url(), status: r.status() }); });
  page.on('requestfailed', r => s.failures.push({ method: r.method(), url: r.url(), reason: r.failure()?.errorText }));
  if (clipboard) await page.addInitScript(() => {
    Object.defineProperty(navigator, 'clipboard', { configurable: true, value: { writeText: text => { window.probeCopied = text; return new Promise((resolve, reject) => { window.probeClipboardResolve = resolve; window.probeClipboardReject = () => reject(new Error('synthetic clipboard denial')); }); } } });
  });
  await page.route('http://localhost:5000/**', async route => {
    const r = route.request(), path = new URL(r.url()).pathname, method = r.method();
    s.requests.push({ path, method });
    const headers = { 'access-control-allow-origin': 'http://localhost:3000', 'access-control-allow-credentials': 'true', 'access-control-allow-methods': 'GET,POST,PATCH,DELETE', 'access-control-allow-headers': 'content-type' };
    const reply = (body, status = 200) => route.fulfill({ status, headers, contentType: 'application/json', body: JSON.stringify(body) });
    if (method === 'OPTIONS') return route.fulfill({ status: 204, headers });
    if (path === '/api/v1/users/me' && method === 'GET') return reply(owner);
    if (path === `/api/v1/forms/${id}` && method === 'GET') return reply(s.dto);
    if (path.endsWith('/result-visibility') && method === 'PATCH') {

      s.dto = { ...s.dto, resultVisibility: JSON.parse(r.postData()).resultVisibility }; return reply(s.dto);

      if (s.writeMode === 'unknown') return route.abort('failed');

    }

  });

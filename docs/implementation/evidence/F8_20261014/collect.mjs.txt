const hash=b=>crypto.createHash('sha256').update(b).digest('hex');

function copy(source,name,normalize=false){const original=fs.readFileSync(source);let result=original;if(normalize){const text=original.toString('utf8').replace(/^\uFEFF/,'').split(/\r?\n/).map(l=>l.replace(/[\t ]+$/,'')).join('\n').replace(/\n*$/,'\n');safeText(text,name);result=Buffer.from(text);}else if(/\.(json|md|txt)$/.test(name))safeText(original.toString('utf8'),name);const target=path.join(out,name);fs.mkdirSync(path.dirname(target),{recursive:true});fs.writeFileSync(target,result);copied.push({path:name,sha256:hash(result),rawPath:path.relative(cwd,source).replaceAll('\\','/'),rawSha256:hash(original),normalized:normalize});}
function inline(body,name,source){const bytes=Buffer.from(body,'base64');const target=path.join(out,name);fs.mkdirSync(path.dirname(target),{recursive:true});fs.writeFileSync(target,bytes);copied.push({path:name,sha256:hash(bytes),rawPath:source,rawSha256:hash(bytes),normalized:false});}

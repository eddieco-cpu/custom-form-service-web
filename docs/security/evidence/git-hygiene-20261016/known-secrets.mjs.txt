import fs from 'node:fs'
import { execFileSync } from 'node:child_process'
const dir = 'output/hygiene-audit-20261016'
const inventory = JSON.parse(fs.readFileSync(`${dir}/inventory.json`, 'utf8'))
const privateFiles = inventory.ignoredLocalScanned.filter(p => /(?:private-credentials|private-state|storageState|storage-state).*\.json$/i.test(p))
const secrets = new Map()
const privateSummary = []
for (const file of privateFiles) {
  let found = 0
  const visit = (value, key = '') => {
    if (typeof value === 'string' && /^(?:password|passwd|token|access_token|accessToken|signingKey|clientSecret)$/i.test(key) && value.length >= 8) {
      found++; if (!secrets.has(value)) secrets.set(value, `LOCAL-SECRET-${secrets.size + 1}`)
    }
    if (Array.isArray(value)) for (const item of value) visit(item)
    else if (value && typeof value === 'object') for (const [k, v] of Object.entries(value)) visit(v, k)
  }
  try { visit(JSON.parse(fs.readFileSync(file, 'utf8'))) }
  catch { privateSummary.push({ path: file, parseError: true }); continue }
  privateSummary.push({ path: file, secretFields: found })
}
const hits = []
function check(buf, meta) {
  if (buf.includes(0)) return
  const content = buf.toString('utf8')
  for (const [value, id] of secrets) {
    let index = content.indexOf(value)
    while (index >= 0) {
      hits.push({ ...meta, id, line: content.slice(0, index).split('\n').length })
      index = content.indexOf(value, index + value.length)
    }
  }
}
for (const file of inventory.tracked) check(fs.readFileSync(file), { scope: 'tracked', path: file })
const objects = execFileSync('git', ['rev-list', '--objects', '--all', '--reflog'], { encoding: 'utf8' }).trim().split('\n').map(l => ({ oid: l.split(' ')[0], path: l.slice(l.indexOf(' ') + 1) }))
const batch = execFileSync('git', ['cat-file', '--batch'], { input: objects.map(o => o.oid).join('\n') + '\n', maxBuffer: 256 * 1024 * 1024 })
let offset = 0
for (const object of objects) {
  const end = batch.indexOf(10, offset); const [, type, size] = batch.subarray(offset, end).toString('utf8').split(' ')
  const buf = batch.subarray(end + 1, end + 1 + Number(size)); offset = end + 2 + Number(size)
  if (type === 'blob') check(buf, { scope: 'history', ...object })
}
const result = { uniqueSecretCount: secrets.size, privateSummary, hits }
fs.writeFileSync(`${dir}/known-secret-results.json`, JSON.stringify(result, null, 2) + '\n')
console.log(JSON.stringify(result, null, 2))

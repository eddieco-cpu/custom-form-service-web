import {test,expect} from '@playwright/test'
const api='http://localhost:5000/api/v1',uid=n=>`00000000-0000-4000-8000-${String(n).padStart(12,'0')}`,fid=uid(900),headers={'access-control-allow-origin':'http://localhost:3000','access-control-allow-credentials':'true'}
test('S07 fixed large fixture all sizes and exact text order/whitespace/Unicode',async({page},info)=>{
  const person=n=>({id:uid(1000+n),displayName:'長'.repeat(90)+String(n).padStart(3,'0')})
  const q=(i,count,statistics,type='text')=>({questionId:uid(100+i),title:i===29?'末題'+ '長'.repeat(2000):'題目'+(i+1),description:i===0?'第一行\n第二行':null,type,count,statistics})
  const groups=['Z','Apple','apple','e\u0301','é','中  間\n換行'].map((value,i)=>({value,count:i===0?2:1,respondents:i===0?[person(1),person(2)]:[person(i+2)]}))
  const dto={formId:fid,title:'固定大型統計',description:'純文字 <script>window.f10Injected=1</script>',creatorDisplayName:'合成建立者',responseMode:'multiple',resultVisibility:'public',submissionDeadlineAt:null,count:200,questions:[q(0,200,Array.from({length:200},(_,i)=>({value:'完整答案 '+String(i+1).padStart(3,'0'),count:1,respondents:[person(i+1)]}))),q(1,0,Array.from({length:101},(_,value)=>({value,count:0,respondents:[]})),'rating'),q(2,120,[{value:'長'.repeat(2000),count:120,respondents:Array.from({length:120},(_,i)=>person(i+1))}]),q(3,7,groups),...Array.from({length:26},(_,i)=>q(i+4,0,[]))]}
  let calls=0;const errors=[];page.on('pageerror',e=>errors.push(e.message))
  await page.route(api+'/**',async r=>{const req=r.request(),path=new URL(req.url()).pathname;if(req.method()==='OPTIONS'){await r.fulfill({status:204,headers});return}if(path==='/api/v1/users/me'){await r.fulfill({status:401,headers,json:{error:{code:'AUTH_REQUIRED',message:'合成guest'}}});return}expect(req.method()).toBe('GET');expect(req.url()).toBe(`${api}/forms/${fid}/statistics`);calls++;await r.fulfill({status:200,headers,json:dto})})
  for(const width of [1440,390,320]){await page.setViewportSize({width,height:width===320?740:900});await page.goto(`/forms/${fid}/statistics`);const qs=page.locator('[data-statistics-question]');await expect(qs).toHaveCount(30);await expect(qs.nth(0).locator('[data-statistics-item]')).toHaveCount(200);await expect(qs.nth(0).locator('[data-statistics-value]').last()).toHaveText('完整答案 200');await expect(qs.nth(1).locator('[data-statistics-value]')).toHaveText(Array.from({length:101},(_,i)=>String(i)));await expect(qs.nth(2).locator('[data-statistics-respondent]')).toHaveCount(120);await expect(qs.nth(2).locator('[data-statistics-value]')).toHaveText('長'.repeat(2000));expect(await qs.nth(3).locator('[data-statistics-value]').allTextContents()).toEqual(groups.map(x=>x.value));await qs.last().scrollIntoViewIfNeeded();await expect(qs.last()).toBeInViewport();expect(await page.evaluate(()=>document.documentElement.scrollWidth<=innerWidth)).toBe(true);expect(await page.evaluate(()=>'f10Injected' in window)).toBe(false);await info.attach(`S07-${width}.png`,{body:await page.screenshot(),contentType:'image/png'})}
  expect(calls).toBe(3);expect(errors).toEqual([]);await info.attach('S07-fixed-independent-fixture',{body:JSON.stringify({simulated:true,expectedTextOrder:groups.map(x=>x.value),questions:30,items:200,respondents:120,longTextCodePoints:2000,widths:[1440,390,320],requestCount:calls}),contentType:'application/json'})
})

import {test,expect} from '@playwright/test'
import {readFile,writeFile} from 'node:fs/promises'
const dir='output/playwright/v1-acceptance/v1-f10-20261016-01',api='http://localhost:5000/api/v1'
test('A01 A09 R13 actual Cookie header flags and same-label Preview uses saved IDs',async({browser})=>{
  const data=JSON.parse(await readFile(dir+'/private-state.json','utf8')),context=await browser.newContext(),p=await context.newPage(),out={realApi:true,interception:false,cookieHeaders:[],reads:[],requests:[],pageErrors:[]}
  p.on('pageerror',e=>out.pageErrors.push(e.message));p.on('request',r=>{if(r.url().startsWith(api+'/'))out.requests.push({method:r.method(),url:r.url(),...(r.url().includes('/forms/')?{body:r.postData()}: {})})})
  const cookieFlags=async response=>{const header=(await response.allHeaders())['set-cookie']??'';return{present:!!header,name:header.startsWith('access_token=')?'access_token':'unexpected',httpOnly:/;\s*httponly/i.test(header),sameSiteLax:/;\s*samesite=lax/i.test(header),pathRoot:/;\s*path=\/(?:;|$)/i.test(header),expires:/;\s*expires=/i.test(header),maxAge:/;\s*max-age=/i.test(header),domain:/;\s*domain=/i.test(header),secure:/;\s*secure(?:;|$)/i.test(header)}}
  const fetchApi=async(path,method='GET',body,expected=200)=>{const result=await p.evaluate(async({url,method,body})=>{const r=await fetch(url,{method,credentials:'include',...(body?{headers:{'content-type':'application/json'},body:JSON.stringify(body)}:{})});return{status:r.status,body:await r.text()}},{url:api+path,method,body});out.reads.push({path,method,...result});expect(result.status).toBe(expected);return result.body?JSON.parse(result.body):null}
  try{
    await p.goto('/login');await p.getByRole('textbox',{name:'Email'}).fill(data.accounts.A.email);await p.getByLabel('密碼',{exact:true}).fill(data.accounts.A.password);const login=p.waitForResponse(r=>r.url()===api+'/auth/login'&&r.request().method()==='POST');await p.getByRole('button',{name:'登入',exact:true}).click();const r=await login;expect(r.status()).toBe(200);const flags=await cookieFlags(r);out.cookieHeaders.push({operation:'login',...flags});expect(flags).toEqual({present:true,name:'access_token',httpOnly:true,sameSiteLax:true,pathRoot:true,expires:true,maxAge:false,domain:false,secure:false});await expect(p).toHaveURL('http://localhost:3000/')
    const f=data.forms['D05-RECHECK'],before=await fetchApi(`/forms/${f.id}`);expect(before.userId).toBe(data.accounts.A.id);expect(before.status).toBe('draft');const q=before.questions.find(q=>q.type==='single_choice');expect(q.options).toHaveLength(0)
    await fetchApi(`/forms/${f.id}`,'PATCH',{questions:[{id:q.id,options:[{label:'同標籤',value:'same_label_A',sortOrder:0},{label:'同標籤',value:'same_label_B',sortOrder:1}]}]});const saved=await fetchApi(`/forms/${f.id}`),options=saved.questions.find(v=>v.id===q.id).options;expect(options.map(o=>o.label)).toEqual(['同標籤','同標籤']);expect(new Set(options.map(o=>o.id)).size).toBe(2)
    await p.goto(`/forms/${f.id}/edit`);await p.getByRole('button',{name:'預覽',exact:true}).click();const dialog=p.getByRole('dialog'),radio=dialog.getByRole('radio',{name:'同標籤',exact:true});await expect(radio).toHaveCount(2);await radio.nth(1).check();await expect(radio.nth(1)).toBeChecked();expect(await radio.nth(1).inputValue()).toBe(options[1].id);await radio.nth(0).check();expect(await dialog.locator('input[type=radio]:checked').count()).toBe(1);expect(await radio.nth(0).inputValue()).toBe(options[0].id);out.sameLabel={questionId:q.id,options,selectedServerIds:options.map(o=>o.id),zeroResponsePost:!out.requests.some(r=>r.method==='POST'&&r.url.endsWith('/responses'))};await p.getByRole('button',{name:'關閉預覽',exact:true}).click();await expect(p.getByRole('button',{name:'保存',exact:true})).toBeDisabled()
    await p.getByRole('button',{name:'使用者選單'}).click();const logout=p.waitForResponse(r=>r.url()===api+'/auth/logout'&&r.request().method()==='POST');await p.getByRole('button',{name:'登出',exact:true}).click();const l=await logout;expect(l.status()).toBe(204);out.cookieHeaders.push({operation:'logout',...await cookieFlags(l)});await expect(p).toHaveURL('http://localhost:3000/login');expect((await context.cookies()).some(c=>c.name==='access_token')).toBe(false);expect((await fetchApi('/users/me','GET',undefined,401)).error.code).toBe('AUTH_REQUIRED');await fetchApi('/auth/logout','POST',undefined,204);expect(out.pageErrors).toEqual([])
  }finally{await writeFile(dir+'/raw/protocol-results.json',JSON.stringify(out,null,2));await context.close()}
})

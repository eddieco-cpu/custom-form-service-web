import {test,expect} from '@playwright/test'
import {readFile,writeFile,mkdir} from 'node:fs/promises'
const dir='output/playwright/v1-acceptance/v1-f10-20261016-01',api='http://localhost:5000/api/v1',origin='http://localhost:3000'
test('E04 A07 real Browser read-only layout, Header, public and production boundaries',async({browser})=>{
  const data=JSON.parse(await readFile(dir+'/private-state.json','utf8')),account=data.accounts.A,ctx=await browser.newContext({viewport:{width:1440,height:900}}),p=await ctx.newPage(),out={method:'real Browser/API reads; UI Login only; no Form writes',browser:browser.version(),layouts:[],requests:[],errors:[],external:[]}
  p.on('pageerror',e=>out.errors.push(e.message));p.on('request',r=>{const u=new URL(r.url());if(u.origin==='http://localhost:5000')out.requests.push({method:r.method(),url:r.url()});else if(![origin,'data:','blob:'].includes(u.origin)&&!['data:','blob:'].includes(u.protocol))out.external.push(r.url())})
  const done=p.waitForResponse(r=>r.url()===api+'/auth/login'&&r.request().method()==='POST');await p.goto('/login');await p.getByRole('textbox',{name:'Email'}).fill(account.email);await p.getByLabel('密碼',{exact:true}).fill(account.password);await p.getByRole('button',{name:'登入',exact:true}).click();expect((await done).status()).toBe(200);await expect(p).toHaveURL(origin+'/')
  const pages=[['home','/',1280],['list','/forms',1280],['profile','/profile',640],['editor',`/forms/${data.forms['D05-RECHECK'].id}/edit`,960],['form',`/forms/${data.forms.EDIT.id}`,800],['statistics',`/forms/${data.forms['ST-I'].id}/statistics`,800]]
  await mkdir(dir+'/raw/screenshots',{recursive:true})
  try{
    for(const width of [1440,390,320])for(const [name,path,max]of pages){
      await p.setViewportSize({width,height:width===320?740:900});await p.goto(path);const section=p.locator(name==='home'?'main':'main > section').first();await expect(section).toBeVisible();if(name==='statistics')await expect(p.locator('[data-statistics-count]')).toBeVisible();if(name==='list')await expect(p.getByRole('list',{name:'我的表單卡片'}).or(p.getByRole('table',{name:'我的表單'})).first()).toBeAttached()
      const layout=await section.evaluate(e=>{const r=e.getBoundingClientRect(),m=document.querySelector('main').getBoundingClientRect(),h=document.querySelector('header > div').getBoundingClientRect();return{sectionWidth:r.width,left:r.left+(e.tagName==='MAIN'?parseFloat(getComputedStyle(e).paddingLeft):0),mainWidth:m.width,headerHeight:h.height,scrollWidth:document.documentElement.scrollWidth,viewport:innerWidth}});out.layouts.push({name,width,expectedMax:max,...layout});expect(layout.sectionWidth).toBeLessThanOrEqual(max+1);expect(layout.mainWidth).toBeLessThanOrEqual(1281);expect(layout.scrollWidth).toBeLessThanOrEqual(width+1);expect(layout.headerHeight).toBe(width<1024?56:64);if(width<1024)expect(layout.left).toBeGreaterThanOrEqual(15)
      if((width===1440&&name==='editor')||(width===390&&['list','statistics'].includes(name))||(width===320&&name==='form'))await p.screenshot({path:dir+`/raw/screenshots/${name}-${width}.png`,fullPage:true})
    }
    for(const width of [1023,1024]){await p.setViewportSize({width,height:900});await p.goto('/');const nav=p.getByRole('navigation',{name:'主要導覽',exact:true});expect(await p.locator('header > div').evaluate(e=>e.getBoundingClientRect().height)).toBe(width===1023?56:64);if(width===1024){await expect(nav).toBeVisible();expect(await nav.getByRole('button').allTextContents()).toEqual(['建立表單','我的表單','個人資訊','首頁'])}else{await expect(nav).toBeHidden();await p.getByRole('button',{name:'展開導覽',exact:true}).focus();await p.keyboard.press('Enter');await expect(p.getByRole('navigation',{name:'手機導覽'})).toBeVisible();await p.getByRole('navigation',{name:'手機導覽'}).getByRole('button',{name:'個人資訊',exact:true}).click();await expect(p).toHaveURL(origin+'/profile');await expect(p.getByRole('navigation',{name:'手機導覽'})).toHaveCount(0)}}
    const g=await browser.newContext({viewport:{width:1440,height:900}}),gp=await g.newPage();await gp.goto('/login');await expect(gp.getByRole('tab',{name:'登入',exact:true})).toBeVisible();const box=await gp.locator('main > section').boundingBox();expect(box.width).toBeLessThanOrEqual(441);expect(await gp.getByRole('button',{name:/Google/i}).count()).toBe(0);await gp.goto(`/forms/${data.forms['ST-A'].id}/statistics`);await expect(gp.locator('[data-statistics-count]')).toBeVisible();await expect(gp.getByRole('button',{name:'使用者選單'})).toHaveCount(0);await gp.getByRole('navigation',{name:'主要導覽',exact:true}).getByRole('button',{name:'建立表單',exact:true}).click();await expect(gp).toHaveURL(/\/login\?redirect=/);expect(new URL(gp.url()).searchParams.get('redirect')).toBe('/forms/new');await g.close()
    expect(out.errors).toEqual([]);expect(out.external).toEqual([]);expect(out.requests.filter(r=>r.method!=='GET'&&!r.url.endsWith('/auth/login'))).toEqual([])
  }finally{await writeFile(dir+'/raw/ui-layout-results.json',JSON.stringify(out,null,2));await ctx.close()}
})

  async function checkpoint(id, label, fn) {
    current = `${id}:${label}`
    const startedAt = new Date().toISOString()
    try { const value = await fn(); state.checkpoints.push({ id, label, status: 'PASS', startedAt, finishedAt: new Date().toISOString() }); console.log(`CHECKPOINT PASS ${current}`); return value }
    catch (e) { state.checkpoints.push({ id, label, status: 'FAIL', detail: clean(e.message), startedAt, finishedAt: new Date().toISOString() }); console.log(`CHECKPOINT FAIL ${current}: ${clean(e.message).slice(0,400)}`); return undefined }
    finally { await flush() }
  }
  const allow = (method, path, status) => { if (status >= 400) state.expectedErrors.push({ case: current, method, url: api + path, status }) }
  function observe(actor) {
    const p = actor.page
    p.on('framenavigated', f => { if (f === p.mainFrame()) state.routes.push({ case: current, actor: actor.alias, url: f.url() }) })
    p.on('pageerror', e => state.pageErrors.push({ case: current, actor: actor.alias, message: clean(e.message) }))
    p.on('console', m => { if (m.type() === 'error' || m.text().includes('[Vue warn]')) state.diagnostics.push({ case: current, actor: actor.alias, text: clean(m.text()), url: m.location().url }) })
    p.on('request', req => {
      if (!req.url().startsWith(api + '/')) return
      const t = { sequence: ++sequence, case: current, actor: actor.alias, method: req.method(), url: req.url(), ...(new URL(req.url()).pathname.startsWith('/api/v1/forms') ? { body: req.postData() } : {}) }
      tickets.set(req, t); state.requests.push(t)
      if (req.method() === 'DELETE') { const id = new URL(req.url()).pathname.split('/').at(-1); if (!Object.entries(privateState.forms).some(([alias, f]) => allowedDelete.has(alias) && f.id === id)) state.deleteViolations.push({ case: current, id }) }
    })
    p.on('response', res => {
      if (!res.url().startsWith(api + '/')) return
      const t = tickets.get(res.request()) ?? { case: current, actor: actor.alias, method: res.request().method(), url: res.url() }
      pending.push((async () => { const h = await res.allHeaders(); const r = { ...t, status: res.status(), origin: h['access-control-allow-origin'] ?? '', credentials: h['access-control-allow-credentials'] ?? '' }; if (new URL(res.url()).pathname.startsWith('/api/v1/forms')) r.response = await res.text().catch(() => '<body unavailable>'); state.outcomes.push(r) })())
    })
  }

    const v = privateState.accounts[a.alias]; secrets.push(v.password)
    if (new URL(a.page.url()).pathname !== '/login' || new URL(a.page.url()).searchParams.get('redirect') !== target) await navigate(a, `/login?redirect=${encodeURIComponent(target)}`)
    await a.page.getByRole('textbox', { name: 'Email' }).fill(v.email)
    await a.page.getByLabel('密碼', { exact: true }).fill(v.password)
    const r = await response(a, '/auth/login', 'POST', () => a.page.getByRole('button', { name: '登入', exact: true }).click())

    const cookie = (await a.context.cookies()).find(c => c.name === 'access_token')

    if (status === 'closed') { await call(a, `/forms/${dto.id}/close`, 'POST'); privateState.forms[alias].status = 'closed' }
    return parsed(await call(a, `/forms/${dto.id}`, 'GET', undefined, 200, `${alias} independent readback`))
